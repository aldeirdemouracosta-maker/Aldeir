cmake_minimum_required(VERSION 3.16)
project(vmfaceswap_pc CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_library(vmfaceswap SHARED vm_c_api.cpp ../../core/src/face_swap.cpp)
target_include_directories(vmfaceswap PRIVATE ../../core/include)
set_target_properties(vmfaceswap PROPERTIES CXX_VISIBILITY_PRESET hidden)
