cmake_minimum_required(VERSION 3.16)
project(vmcore_host_tests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vmcore STATIC src/timeline.cpp src/pixel_ops.cpp)
target_include_directories(vmcore PUBLIC include)
target_compile_options(vmcore PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests vmcore)
add_test(NAME core_tests COMMAND core_tests)
